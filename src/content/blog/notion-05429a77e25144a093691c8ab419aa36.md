반복문은 collection 타입과 많이 사용이 된다\.

#### for\-in 구문

```swift
for item in items {
    /* 실행 구문 */
}

var integers = [1, 2, 3]
let people = ["yagom": 10, "eric": 15, "mike": 12]

for integer in integers {
    print(integer)
}

// Dictionary의 item은 key와 value로 구성된 튜플 타입
for (name, age) in people {
    print("\(name): \(age)")
}

for i in stride(from : 0, to : 5, by : 1) {
	print("\(i)")
}
```

#### while 구문

```swift
while 조건 {
    /* 실행 구문 */
}

while integers.count > 1 {
    integers.removeLast()
}
```

#### repeat\-while 구문

기존 언어의 do\-while 구문과 유사하다\.

```swift
repeat {
    /* 실행 구문 */
} while 조건

repeat {
    integers.removeLast()
} while integers.count > 0
```
