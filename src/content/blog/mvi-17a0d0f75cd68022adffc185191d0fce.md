블로그에서 본대로 순서를 지켜가며 작성 하였습니다\. 

#### 1\. ModelStateProtocol

```swift
/// View에서 보여지는 모든 Model(= State) 을 선언 하는 ModelStateProtocol
/// Model에서 VIew를 그릴 때, 내부 로직은 숨기고 Property에 대한 접근만 허용
protocol ModelStateProtocol {
    var text: String { get }
    var retryButton: String { get }
}

```

#### 2\. ModelActionProtocol

```swift
/// View에서 보여지는 모든 Model을 가공하기 위한 ModelActionProtocol
/// Model 내의 상태 값 변경, UI 적용을 위한 데이터 가공
/// Intent 를 통해서 호출
protocol ModelActionProtocol: AnyObject {
    func parse(number: Int)
}
```

#### 3\. Model

```swift
import SwiftUI
/// ModelStateProtocol, ModelActionProtocol을 상속 받아서 구현한 모델

/// View에서 보여질 State 값 구현
final class Model: ObservableObject, ModelStateProtocol{
    
    // 변경될 경우 바로 View에 전달되어 업데이트가 되어야 하기 때문에 ObservableObject, @Published 선언
    @Published var text: String = ""
    
    @Published var retryButton: String = "retry Button"
}

/// Intent로부터 데이터나 이벤트를 받아서 처리할 로직 구현
extension Model: ModelActionProtocol{
    func parse(number: Int) {
        text = "Random number: \(number)"
    }
}
```

#### 4\. IntentProtocol

```swift
/// View에서 event를 받아오기 위한 Protocol
/// View에서 발생할 event에 대한 함수 정의
protocol IntentProtocol {
    func viewOnAppear()
    func tappedButton()
}
```

#### 5\. Intent

```swift
/// View에서 이벤트가 들어올 경우에 실행할 비즈니스 로직 정의
class Intent {
    
    // Intent에서는 Model의 State에는 접근하지 않고 Action에만 접근함.
    private weak var model: ModelActionProtocol?
    
    init(model: ModelActionProtocol) {
        self.model = model
    }
}


// 즉, Model의 Action에서 Model의 State값에 할당해주는 로직을 구현 하는 거임.
// 서버의 통신이 있는 경우 Intent에서 해당 로직을 실행하고 그 결과에 model에 어떤 action을 처리하라고 전달.
extension Intent: IntentProtocol {
    func viewOnAppear() {
        let number = Int.random(in: 0 ..< 50)
        model?.parse(number: number)
    }
    
    func tappedButton() {
        let number = Int.random(in: 0 ..< 50)
        model?.parse(number: number)
    }
}
```

<strong>weak를 사용한 이유? 그리고 ModelActionProtocol이 AnyObject를 상속받은 이유?</strong> 

weak를 사용한 이유는 MVI 패턴은 Model → View → Intent → Model 순으로 참조가 이어지는데 이때 Intent가 Model을 강한 참조하게 되면 순환 참조가 발생할 수 있다\.

→ 대충 순환하는 구조이기 때문에 약한 참조를 해줘야 한다는 것

weak 키워드는 클래스 타입에만 사용할 수 있고, AnyObject를 채택함으로써 이 프로토콜은 클래스에서만 사용될 수 있음을 명시한다\. 

#### 6\. MVIContainer

```swift
import SwiftUI
import Combine

/// Intent와 Model을 받는 클래스, Intent와 Model에 대한 참조를 구현
final class MVIContainer<Intent, Model>: ObservableObject{
    let intent: Intent
    let model: Model
    
    private var cancellables: Set<AnyCancellable> = []
    
    // objectWillChange를 통해 Model의 변화를 감지하고 변화가 감지되면 Container가 아닌 View로 연결
    init(intent: Intent, model: Model, modelChangedPublisher: ObjectWillChangePublisher){
        self.intent = intent
        self.model = model
        
        modelChangedPublisher
            .receive(on: RunLoop.main)
            .sink(receiveValue: objectWillChange.send)
            .store(in: &cancellables)
    }
}
```

#### 7\. View, Build Function

```swift
import SwiftUI

struct ContentView: View {
    // View가 다시 생성될 때 Intent와 Model은 다시 생성하지 않도록 하기 위해
    @StateObject var container: MVIContainer<Intent, ModelStateProtocol>
    private var intent: Intent { container.intent }
    
    // 다시 한번 말하지만 View에서는 Model의 Action이 노출되서는 안된다. 절대.
    private var state: ModelStateProtocol { container.model }
    
    var body: some View {
        VStack {
            Text(state.text)
                .onAppear {
                    container.intent.viewOnAppear()
                }
            Button(state.retryButton){
                self.intent.tappedButton()
            }
        }
    }
}

extension ContentView {
    // model과 intent를 묶은 container를 포함하는 View를 생성하여 리턴해주는 함수
    static func build() -> some View {
        let model = Model()
        let intent = Intent(model: model)
        let container = MVIContainer(intent: intent, model: model as ModelStateProtocol, modelChangedPublisher: model.objectWillChange)
        
        let view = ContentView(container: container)
        
        return view
    }
}
```
