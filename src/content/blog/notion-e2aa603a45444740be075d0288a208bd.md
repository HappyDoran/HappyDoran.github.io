#### Array \(배열\)

순서가 있는\(인덱스 존재\) 리스트 컬렉션

```swift
var integers : Array<Int> = Array<Int>()
// var integers : Array<Int> = [Int]()
// var integers : [Int] = [Int]()
// var integers : [Int] = []

integers.append(1)
integers.append(100)
//[1,100]

integers.contains(100)
//true

integers.remove(at:0)
//0번째에 있는 인덱스를 없애달라

integers.removeLast()
integers.removeFirst()

integers.count //배열 갯수
```

#### Dictionary 

키와 값의 쌍으로 이루어진 컬렉션

```swift
var anyDictionary : Dictionary<String, Any> = [String : Any]()

anyDictionary["someKey"] = "someValue"
anyDictionary["anotherKey"] = 100
//["someKey" : "someValue", "anotherKey":100]

anyDictionary["someKey"] = "dictionary"
//["someKey" : "dictionary", "anotherKey":100]

anyDictionary.removeValue(forKey : "anotherKey")
//["someKey" : "someValue"]

anyDictionary["someKey"] = nil
//removeValue와 같은 작동
```

#### Set \(집합\)

순서가 없고, 멤버가 유일한 컬렉션

```swift
var integerSet : Set<Int> = Set<Int>()

integerSet.insert(1)
integerSet.insert(100)
integerSet.insert(99)
integerSet.insert(99)
//{100,99,1}

integerSet.contains(1)
//true
integerSet.contains(2)
//false

integerSet.remove(100)
//{99,1}
integerSet.removeFirst()
//{1}

let setA : set<Int> = [1, 2, 3, 4, 5]
let setB : set<Int> = [3, 4, 5, 6, 7]

let union : set<Int> = setA.union(setB)
//{2,4,5,6,7,3,1}

let sortedunion : [Int] = union.sorted()
//{1,2,3,4,5,6,7] -- 정렬된 순서의 배열

let intersetion : Set<Int> = setA.intersection(setB)
//{5,3,4} -- 교집합

let subtraction : Set<Int> = setA.subtracting(setB)
//{2,1} -- 차집합
```
